cmake_minimum_required(VERSION 3.22)
project(Scintillua VERSION 6.7 LANGUAGES CXX)
find_package(Scintilla CONFIG REQUIRED)
find_package(Lexilla CONFIG REQUIRED)
find_package(Lua REQUIRED)
find_package(unofficial-lpeg CONFIG REQUIRED)
add_library(Scintillua SHARED Scintillua.cxx)
target_compile_features(Scintillua PUBLIC cxx_std_17)
target_link_libraries(Scintillua PUBLIC Scintilla::Headers PRIVATE Lexilla::Lexilla ${LUA_LIBRARIES} unofficial::lpeg::lpeg)
target_include_directories(Scintillua PRIVATE ${LUA_INCLUDE_DIR} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:include/scintillua>)
install(TARGETS Scintillua EXPORT ScintilluaTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES Scintillua.h DESTINATION include/scintillua)
install(DIRECTORY lexers/ DESTINATION share/scintillua/lexers FILES_MATCHING PATTERN "*.lua")
include(CMakePackageConfigHelpers)
configure_package_config_file(ScintilluaConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ScintilluaConfig.cmake" INSTALL_DESTINATION lib/cmake/Scintillua)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ScintilluaConfigVersion.cmake" VERSION 6.7 COMPATIBILITY SameMajorVersion)
install(EXPORT ScintilluaTargets NAMESPACE Scintillua:: DESTINATION lib/cmake/Scintillua)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ScintilluaConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ScintilluaConfigVersion.cmake" DESTINATION lib/cmake/Scintillua)
