cmake_minimum_required(VERSION 3.25)
project(cpython_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
find_package(cpython CONFIG REQUIRED)
enable_testing()

add_executable(cpython_embed embed.cpp)
target_link_libraries(cpython_embed PRIVATE CPython::Python)
add_library(sdk_cpython_extension MODULE extension.cpp)
target_link_libraries(sdk_cpython_extension PRIVATE CPython::Python)
set_target_properties(sdk_cpython_extension PROPERTIES PREFIX "")
if(WIN32)
    set_target_properties(sdk_cpython_extension PROPERTIES SUFFIX ".pyd" DEBUG_POSTFIX "_d")
    set(interpreter "${SDK_PYTHON_PREFIX}/tools/python3/$<IF:$<CONFIG:Debug>,debug/python_d.exe,python.exe>")
else()
    set(interpreter "${SDK_PYTHON_PREFIX}/tools/python3/$<IF:$<CONFIG:Debug>,debug/python3.15td,python3.15t>")
endif()
add_test(NAME cpython_interpreter COMMAND "${interpreter}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/check.py" "$<TARGET_FILE_DIR:sdk_cpython_extension>")
add_test(NAME cpython_embed COMMAND cpython_embed
    "${CMAKE_CURRENT_SOURCE_DIR}/check.py" "$<TARGET_FILE_DIR:sdk_cpython_extension>")
set_tests_properties(cpython_interpreter cpython_embed PROPERTIES TIMEOUT 90)
