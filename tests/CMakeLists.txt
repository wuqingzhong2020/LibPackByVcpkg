cmake_minimum_required(VERSION 3.25)
project(cpp20_sdk_validation LANGUAGES C CXX)
if(NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 20)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
    add_compile_options(/utf-8 /Zc:__cplusplus /bigobj)
endif()
add_compile_definitions(EIGEN_MPL2_ONLY)
enable_testing()
option(SDK_FULL "Validate all custom ports as well as native packages" ON)
add_library(sdk_runtime_audit OBJECT runtime-audit.cpp)
function(sdk_test name)
    add_executable(${name} consumers/${name}.cpp)
    target_link_libraries(${name} PRIVATE sdk_runtime_audit ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 90)
endfunction()
add_subdirectory(native)
if(SDK_FULL)
    add_subdirectory(custom)
endif()
